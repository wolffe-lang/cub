    // cb18 pickup probe (never committed)
    use forest
    use silhouette
    use sim
    
    fn arena_with(kind: int, seed: int) -> sim.Sim {
        let area = if kind == sim.BOAR {
            forest.BRAMBLEHOLD
        } else if kind == sim.LYNX {
            forest.CRAGS
        } else if kind == sim.ELK {
            forest.RIVER
        } else {
            forest.GRAVEHOLLOW
        }
        var s = sim.new(seed, 80, 24)
        s.heart = 6
        s.fang = 5
        s.health = sim.health_max(6)
        s.gate_open = true
        let (cx, cy, _) = forest.arena(seed, s.field, area)
        let o = forest.origin(seed, area)
        sim.enter_area(mut s, area, cx * sim.FP + sim.FP / 2, (o + cy) * sim.FP + sim.FP / 2)
        // Keep the boss alone.
        var kept = List[sim.Foe]()
        for f in s.foes { if sim.is_boss(f.kind) { (mut kept).push(copy f) } }
        s.foes = kept
        assert(s.foes.len == 1 && s.foes[0].kind == kind, "the area's boss")
        // The cub north of the boss, one clear cell between the bodies (a
        // bite's reach), facing south: its own cell four rows above the boss's
        // top row.
        let (w, h) = silhouette.foe_size(kind)
        let (_, dr, _, _) = silhouette.footprint(w, h, 0)
        s.y = s.foes[0].y + (dr - 4) * sim.FP
        s.x = s.foes[0].x
        s.heading = 128
        s.heading_target = 128
        s.body_facing = 4
        s.foes[0].heading = 0
        sim.settle_bodies(mut s)
        s
    }
    
    fn scenario(ticks: int) {
        // The cub stands outside the arena, three cells past its edge, in the
        // boss's sight; the boss hunts it to the edge and no further, two
        // hundred ticks, in every arena.
        var kind = sim.BOAR
        while kind <= sim.ALPHA {
            var s = arena_with(kind, 1)
            let (cx, cy, r) = forest.arena(s.seed, s.field, s.area)
            let o = forest.origin(s.seed, s.area)
            // Up the route, outside the circle (the boar's clearing: west, at
            // its mouth's row).
            if kind == sim.BOAR {
                s.x = (cx - r - 3) * sim.FP + sim.FP / 2
                s.y = (o + cy) * sim.FP + sim.FP / 2
            } else {
                s.x = forest.spine(s.seed, s.area, cy - r - 3) * sim.FP + sim.FP / 2
                s.y = (o + cy - r - 3) * sim.FP + sim.FP / 2
            }
            s.foes[0].engaged = true
            var n = 0
            while n < ticks {
                sim.step(mut s, List[int]())
                assert(
                    sim.in_arena_at(s, s.foes[0].x, s.foes[0].y),
                    "boss {kind} left its arena at tick {n}: {s.foes[0].x},{s.foes[0].y}",
                )
                assert(!sim.overlap(sim.cub_body(s), sim.foe_body(s.foes[0])), "no overlap {kind} {n}")
                n += 1
            }
            kind += 1
        }
    }
    
    fn main() -> int {
        scenario(20)
        0
    }
